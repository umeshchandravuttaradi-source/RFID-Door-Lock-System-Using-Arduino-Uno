
#include <SPI.h>
#include <MFRC522.h>

#define SS_PIN 10
#define RST_PIN 9

#define GREEN_LED 6
#define RED_LED 7
#define BUZZER 5
#define RELAY 8

MFRC522 rfid(SS_PIN, RST_PIN);

byte authorizedUID[] = {0x23, 0x7A, 0xB1, 0x4C};
byte authorizedUIDSize = 4;

byte unauthorizedUID[] = {0x53, 0x91, 0xA2, 0x18};
byte unauthorizedUIDSize = 4;

#define RELAY_ON LOW
#define RELAY_OFF HIGH

void setup() {
  Serial.begin(9600);

  SPI.begin();
  rfid.PCD_Init();

  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  pinMode(RELAY, OUTPUT);

  lockDoor();

  Serial.println("RFID DOOR SECURITY SYSTEM");
  Serial.println("Door Status: LOCKED");
  Serial.println("Scan RFID card...");
}

void loop() {
  if (!rfid.PICC_IsNewCardPresent()) {
    return;
  }

  if (!rfid.PICC_ReadCardSerial()) {
    return;
  }

  Serial.print("Scanned UID: ");

  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) {
      Serial.print("0");
    }

    Serial.print(rfid.uid.uidByte[i], HEX);
    Serial.print(" ");
  }

  Serial.println();

  if (compareUID(authorizedUID, authorizedUIDSize)) {
    Serial.println("AUTHORIZED CARD");
    Serial.println("ACCESS GRANTED");

    unlockDoor();

    delay(5000);

    lockDoor();

    Serial.println("Door automatically LOCKED");
  }

  else if (compareUID(unauthorizedUID, unauthorizedUIDSize)) {
    Serial.println("UNAUTHORIZED CARD");
    Serial.println("ACCESS DENIED");

    lockDoor();

    for (byte i = 0; i < 3; i++) {
      digitalWrite(BUZZER, HIGH);
      delay(200);
      digitalWrite(BUZZER, LOW);
      delay(200);
    }

    delay(500);
  }

  else {
    Serial.println("UNKNOWN CARD");
    Serial.println("ACCESS DENIED");

    lockDoor();

    for (byte i = 0; i < 4; i++) {
      digitalWrite(BUZZER, HIGH);
      delay(100);
      digitalWrite(BUZZER, LOW);
      delay(100);
    }
  }

  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();

  delay(200);
  rfid.PCD_Init();

  Serial.println("Ready for next card...");
}

bool compareUID(byte storedUID[], byte storedSize) {
  if (rfid.uid.size != storedSize) {
    return false;
  }

  for (byte i = 0; i < storedSize; i++) {
    if (rfid.uid.uidByte[i] != storedUID[i]) {
      return false;
    }
  }

  return true;
}

void unlockDoor() {
  digitalWrite(GREEN_LED, HIGH);
  digitalWrite(RED_LED, LOW);

  digitalWrite(BUZZER, HIGH);
  delay(200);
  digitalWrite(BUZZER, LOW);

  digitalWrite(RELAY, RELAY_ON);
}

void lockDoor() {
  digitalWrite(RELAY, RELAY_OFF);

  digitalWrite(GREEN_LED, LOW);
  digitalWrite(RED_LED, HIGH);
}
